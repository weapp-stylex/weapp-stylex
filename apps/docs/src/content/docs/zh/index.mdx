<Tabs label="框架示例">

<Tab label="styles.ts">

```ts
import * as stylex from 'weapp-stylex'

export const styles = stylex.create({
  card: {
    padding: 16,
    marginTop: '12rpx',
    borderRadius: 12,
    backgroundColor: '#edf7f1',
  },
})
```

</Tab>

<Tab label="原生微信">

```ts
import * as stylex from 'weapp-stylex'
import { styles } from './styles'

Page({
  data: { card: stylex.attrs(styles.card) },
})
```

```xml
<view class="{{card.class}}" style="{{card.style}}">
  StyleX
</view>
```

</Tab>

<Tab label="Wevu">

```vue
<script setup lang="ts">
import * as stylex from 'weapp-stylex'
import { computed } from 'wevu'
import { styles } from './styles'
const card = computed(() => stylex.attrs(styles.card))
</script>

<template>
  <view :class="card.class" :style="card.style">StyleX</view>
</template>
```

</Tab>

<Tab label="Taro React">

```tsx
import { View } from '@tarojs/components'
import * as stylex from 'weapp-stylex'
import { styles } from './styles'

export default function Card() {
  return <View {...stylex.props(styles.card)}>StyleX</View>
}
```

</Tab>

